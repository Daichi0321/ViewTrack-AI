from django import forms class WatchTimeForm(forms.Form): hours = forms.IntegerField(min_value=0, required=False, label="時間") minutes = forms.IntegerField(min_value=0, max_value=59, required=False, label="分") seconds = forms.IntegerField(min_value=0, max_value=59, required=False, label="秒") def clean(self): cleaned = super().clean() h = cleaned.get("hours") or 0 m = cleaned.get("minutes") or 0 s = cleaned.get("seconds") or 0 cleaned["watch_seconds"] = h * 3600 + m * 60 + s return cleaned
